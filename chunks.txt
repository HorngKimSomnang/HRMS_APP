    /**
     * Confirm a pending contract — transitions it to 'active'.
     * Only Super Admin or a Manager whose managed department matches the employee's department.
     */
    public function confirm(Request $request, Contract $contract)
    {
        if ($contract->status !== 'pending') {
            return response()->json(['message' => 'Only pending contracts can be confirmed.'], 422);
        }

        $actor = $request->user();
        $isSuperAdmin = $actor?->hasRole('Super Admin');

        if (!$isSuperAdmin) {
            // Manager must have the employee's department in their managed departments
            $employee    = $contract->employee()->with('department')->first();
            $managedIds  = $actor?->managedDepartments()->pluck('departments.id')->toArray() ?? [];
            if (empty($managedIds) || !in_array($employee?->department_id, $managedIds, true)) {
                return response()->json(['message' => 'You are not authorized to confirm this contract.'], 403);
            }
        }

        $contract->update([
            'status'       => 'active',
            'confirmed_by' => $actor?->id,
            'confirmed_at' => now(),
        ]);

        return response()->json([
            'message' => 'Contract confirmed and activated.',
            'data'    => $contract->fresh(),
        ]);
    }

    /**
     * Count of pending contracts the authenticated user can confirm.
     * Super Admin sees all; Managers see only their department(s).
     */
    public function pendingCount(Request $request)
    {
        $actor = $request->user();
        $query = Contract::where('status', 'pending');

        if (!$actor?->hasRole('Super Admin')) {
            $managedIds = $actor?->managedDepartments()->pluck('departments.id')->toArray() ?? [];
            if (empty($managedIds)) {
                return response()->json(['count' => 0]);
            }
            $query->whereHas('employee', fn($q) => $q->whereIn('department_id', $managedIds));
        }

        return response()->json(['count' => $query->count()]);
    }

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\Contract;
use App\Models\ContractStatusLog;
use App\Services\PayrollService;
use Carbon\Carbon;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class ContractController extends Controller
{
    public function activate(Request $request, Contract $contract)
    {
        if ($contract->status !== 'pending') {
            return response()->json(['message' => 'Contract is not pending.'], 400);
        }

        DB::beginTransaction();
        try {
            $now = Carbon::now();
            
            $contract->update([
                'status' => 'active',
                'start_date' => $now->toDateString(),
                'end_date' => $now->copy()->addMonth()->toDateString(),
            ]);

            ContractStatusLog::create([
                'contract_id' => $contract->id,
                'employee_id' => $contract->employee_id,
                'status' => 'active',
                'changed_at' => $now,
            ]);

            // Now that the contract is active, generate the initial payslip
            $payrollService = app(PayrollService::class);
            $payrollService->generateForEmployee($contract->employee, $now->month, $now->year, true);

            DB::commit();
            return response()->json(['message' => 'Contract activated successfully.', 'contract' => $contract]);
        } catch (\Exception $e) {
            DB::rollBack();
            return response()->json(['message' => 'Failed to activate contract: ' . $e->getMessage()], 500);
        }
    }

    public function destroy(Request $request, Contract $contract)
    {
        DB::beginTransaction();
        try {
            // Log deletion
            ContractStatusLog::create([
                'contract_id' => $contract->id,
                'employee_id' => $contract->employee_id,
                'status' => 'deleted',
                'changed_at' => Carbon::now(),
            ]);

            // Soft-delete contract
            $contract->delete();

            // Hard-delete the current draft payslip
            $contract->employee->payslips()->where('status', 'draft')->forceDelete();

            DB::commit();
            return response()->json(['message' => 'Contract deleted and auto-operations halted.']);
        } catch (\Exception $e) {
            DB::rollBack();
            return response()->json(['message' => 'Failed to delete contract: ' . $e->getMessage()], 500);
        }
    }
}
